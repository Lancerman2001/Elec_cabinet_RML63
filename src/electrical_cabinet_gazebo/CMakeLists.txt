cmake_minimum_required(VERSION 3.16)
project(electrical_cabinet_gazebo)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra)
find_package(ament_cmake REQUIRED)
find_package(gazebo REQUIRED)
find_package(rclcpp REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(cabinet_sim_support REQUIRED)
include_directories(include ${GAZEBO_INCLUDE_DIRS})
link_directories(${GAZEBO_LIBRARY_DIRS})
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/DoorCloseCommand.msg"
  "msg/DoorCloseStatus.msg"
  DEPENDENCIES std_msgs)
rosidl_get_typesupport_target(typesupport_target ${PROJECT_NAME} "rosidl_typesupport_cpp")
add_library(electrical_cabinet_kinematic_plugin SHARED src/electrical_cabinet_kinematic_plugin.cpp src/door_close_state_machine.cpp)
ament_target_dependencies(electrical_cabinet_kinematic_plugin rclcpp gazebo_ros sensor_msgs std_msgs std_srvs cabinet_sim_support)
target_link_libraries(electrical_cabinet_kinematic_plugin ${GAZEBO_LIBRARIES})
target_link_libraries(electrical_cabinet_kinematic_plugin "${typesupport_target}")
install(TARGETS electrical_cabinet_kinematic_plugin LIBRARY DESTINATION lib)
foreach(asset_dir urdf mujoco assets meshes materials config model_mjcf model_source model_gazebo gazebo fonts worlds launch scripts)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${asset_dir}")
    install(DIRECTORY ${asset_dir} DESTINATION share/${PROJECT_NAME}
      PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
  endif()
endforeach()
ament_package()
