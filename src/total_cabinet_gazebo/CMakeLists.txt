cmake_minimum_required(VERSION 3.16)
project(total_cabinet_gazebo)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra)
find_package(ament_cmake REQUIRED)
find_package(gazebo REQUIRED)
find_package(rclcpp REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(cabinet_sim_support REQUIRED)
include_directories(include ${GAZEBO_INCLUDE_DIRS})
link_directories(${GAZEBO_LIBRARY_DIRS})
add_library(cabinet_state_plugin SHARED src/cabinet_state_plugin.cpp)
ament_target_dependencies(cabinet_state_plugin rclcpp gazebo_ros sensor_msgs std_msgs std_srvs cabinet_sim_support)
target_link_libraries(cabinet_state_plugin ${GAZEBO_LIBRARIES})
install(TARGETS cabinet_state_plugin LIBRARY DESTINATION lib)
foreach(asset_dir urdf mujoco assets meshes materials config model_mjcf model_source model_gazebo gazebo fonts worlds launch scripts)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${asset_dir}")
    install(DIRECTORY ${asset_dir} DESTINATION share/${PROJECT_NAME}
      PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
  endif()
endforeach()
install(PROGRAMS scripts/unified_control.py scripts/gazebo_control.py scripts/door_control.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS run_mujoco.sh run_interaction.sh DESTINATION share/${PROJECT_NAME})
install(FILES manifest.json requirements-mujoco.txt DESTINATION share/${PROJECT_NAME})
ament_package()
